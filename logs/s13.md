def count_vowels(s)
    count = 0
    for c in s:
        if c in 'aeiou':
            count += 1 
            return count --> stops the loop right away
        return count
print(count_vowels('apple')) --> 1
print(count_vowels('ski')) --> 0
print(count_vowels('ski')) --> 1
